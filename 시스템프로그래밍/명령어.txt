chmod

chmod 644 ./test.c
//소유자 6(읽기,쓰기), 그룹 4(읽기), 제3자 4(읽기) 가능

gcc
$ gcc -o hw helloworld.c
$ gcc helloworld.c -o hw
//c언어 helloworld.c를 hw 이름으로 컴파일
$ gcc -o helloworld -DMAXLEN=255 helloworld.c
//#define MAXLEN 255와 같음
$ gcc -E -o helloworld.i helloworld.C
//컴파일 과정 중 C preprocessing 까지만 처리하고 나머지는 처리하지 않음
$ gcc -S helloworld.c
//전처리된 파일을 어셈블리 파일로 컴파일까지만 수행, helloworld.s 생성
$ gcc -c helloworld.c
//컴파일 작업만 수행, 오브젝트 파일 helloworld.o 생성
$ gcc -c helloworld.c -linclude
//#include문장에서 지정한 헤더 파일이 들어있는 곳을 정함
$ gcc -g helloworld.C
//디버깅 모드로 컴파일, 오류 발생시 소스코드 단위로 tracing할 수 있도록 추가 데이터 삽입, gdb프로그램으로 디버깅할 때보다 수월하게 디버깅 가능

./helloworld
